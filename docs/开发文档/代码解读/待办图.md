class TodoState(TypedDict):
    messages: Annotated[list, add_messages]
    user_id: int
    thread_id: str
    
    # 意图分析结果

    # 流程控制
    needs_clarification: bool
    needs_confirmation: bool
    resolved_todo_id: Optional[int]

## 节点定义

### analyze_intent 节点

```python
def analyze_intent(state, config):
    """分析用户意图，确定要执行的操作。"""
    messages = state["messages"]
    user_msg = messages[-1].content
    
    # 使用 LLM 分析意图
    result: IntentResult = llm.with_structured_output(IntentResult).invoke(
        [SystemMessage(content=INTENT_PROMPT), HumanMessage(content=user_msg)]
    )
    
    return {
        "detected_intent": result.intent,
        "pending_operation": result.operation,
        "needs_clarification": result.needs_clarification
    }
```

### resolve_entity 节点

```python
def resolve_entity(state, config):
    """解析模糊的待办标识为具体 ID。"""
    operation = state["pending_operation"]
    user_id = get_user_id(state, config)
    
    # 模糊匹配
    matches = fuzzy_match_todos(operation["target"], user_id)
    
    if len(matches) == 0:
        return {"needs_clarification": True, "clarification_reason": "not_found"}
    elif len(matches) == 1:
        return {"resolved_todo_id": matches[0].id}
    else:
        return {"needs_clarification": True, "clarification_reason": "multiple"}
```

## 路由函数

```python
def route_next(state) -> Literal["clarify", "confirm", "execute", "summarize"]:
    if state.get("needs_clarification"):
        return "clarify"
    
    intent = state.get("detected_intent")
    
    # 危险操作需要确认
    if intent in ["delete", "update"] and state.get("needs_confirmation", True):
        return "confirm"
    
    if intent in ["add", "delete", "update", "complete"]:
        return "execute"
    
    return "summarize"
```

## 确认流程

```python
def ask_confirmation(state, config):
    """发送确认请求给用户。"""
    writer = get_stream_writer()
    operation = state["pending_operation"]
    
    # 发送确认事件
    emit_confirmation(writer, operation, "确认执行此操作？")
    
    # 使用 interrupt 等待用户响应
    return interrupt({"action_requests": [operation]})

def wait_for_confirmation(state, config):
    """处理用户确认响应。"""
    # 从 interrupt 恢复时，用户决定在 state 中
    decision = state.get("user_decision", {})
    
    if decision.get("type") == "accept":
        return {"needs_confirmation": False}
    elif decision.get("type") == "reject":
        return {"messages": [AIMessage(content="已取消操作")]}
```
